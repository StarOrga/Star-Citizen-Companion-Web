   ```js
   // icon-controls.mjs – Knöpfe/Links, die nur ein Symbol zeigen, ein aria-label haben, aber keinen [scTooltip].
   import fs from 'node:fs'; import path from 'node:path';
   const ROOT = process.argv[2] ?? 'src/app';
   const walk = (d) => fs.readdirSync(d, { withFileTypes: true }).flatMap((e) => {
     const p = path.join(d, e.name);
     return e.isDirectory() ? walk(p) : p.endsWith('.ts') && !p.endsWith('.spec.ts') ? [p] : [];
   });
   const out = [];
   for (const f of walk(ROOT)) {
     const s = fs.readFileSync(f, 'utf8');
     for (const tag of ['button', 'a']) {           // getrennte Durchläufe: ein <button> in einem <a> wird auch gefunden
       const re = new RegExp(`<${tag}\\b((?:[^<>"]|"[^"]*")*?)>([\\s\\S]*?)<\\/${tag}>`, 'g');
       let m;
       while ((m = re.exec(s))) {
         const [, open, inner] = m;
         if (/scTooltip/.test(open) || !/aria-label/.test(open)) continue;
         if (tag === 'a' && (/<(img|sc-fallback-image|button)\b/.test(inner) || !inner.trim())) continue; // Bild-/Flächenlinks
         const txt = inner.replace(/<!--[\s\S]*?-->/g, '').replace(/&[a-z]+;|&#\d+;/g, '×')
           .replace(/@(if|else|for|switch|case|default)\b[^{]*\{/g, '').replace(/\}/g, '').replace(/<[^>]+>/g, ' ');
         if (/\{\{/.test(txt) || /[A-Za-zÄÖÜäöü]{3,}/.test(txt)) continue;  // sichtbarer Text vorhanden
         const line = s.slice(0, m.index).split('\n').length;
         const cls = (open.match(/class="([^"]+)"/) ?? [, ''])[1];
         out.push(`${f}:${line}  <${tag} class="${cls}">  ${txt.trim().slice(0, 4) || '(svg)'}`);
       }
     }
   }
   out.sort();
   console.log(out.join('\n'));
   console.log(`\n${out.length} Symbol-Bedienelemente mit aria-label, aber ohne App-Tooltip`);
   ```

### Schritt 9 – Guard erweitern

**Dateien:** `scripts/check-native-ui.mjs` (aus D03 Schritt 8; fehlt es, dort beschrieben anlegen).

**Änderung:** Musterliste ergänzen (nur `src/app/**/*.ts` ohne `*.spec.ts`):

`[title]=` bewusst **nicht** prüfen (Component-Inputs wie `sc-app-download-panel`, a11y-Attribut am iframe).
