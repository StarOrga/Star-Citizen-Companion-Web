  // "sicher"     = weder als Literal noch über ein Präfix ('a.b.' + x, `a.b.${x}`) erreichbar UND das
  //                letzte Segment steht nirgends in Anführungszeichen (kein Aufbau aus Einzelteilen möglich).
  // "vielleicht" = nicht direkt referenziert, aber das letzte Segment taucht irgendwo als String auf.
  import fs from 'node:fs'; import path from 'node:path';
  const flat = (o, p = '', out = {}) => { for (const [k, v] of Object.entries(o)) { const kk = p ? `${p}.${k}` : k; if (v && typeof v === 'object') flat(v, kk, out); else out[kk] = v; } return out; };
  const de = flat(JSON.parse(fs.readFileSync('public/i18n/de.json', 'utf8')));
  const walk = (d) => fs.readdirSync(d, { withFileTypes: true }).flatMap((e) => { const p = path.join(d, e.name); return e.isDirectory() ? walk(p) : p.endsWith('.ts') && !p.endsWith('.spec.ts') ? [p] : []; });
  const src = walk('src/app').map((f) => fs.readFileSync(f, 'utf8')).join('\n');
  const exact = new Set(); const prefixes = new Set(); let m;
  const lit = /['"`]([A-Za-z0-9_\-]+(?:\.[A-Za-z0-9_\-]+)+\.?)['"`]/g;
  while ((m = lit.exec(src))) (m[1].endsWith('.') ? prefixes : exact).add(m[1]);
  const tpl = /`([A-Za-z0-9_\-]+(?:\.[A-Za-z0-9_\-]+)*\.)\$\{/g;
  while ((m = tpl.exec(src))) prefixes.add(m[1]);
  const quoted = new Set(); const q = /['"`]([A-Za-z0-9_\-]+)['"`]/g;
  while ((m = q.exec(src))) quoted.add(m[1]);
  const used = (k) => exact.has(k) || [...prefixes].some((p) => k.startsWith(p)) || [...exact].some((e) => k.startsWith(e + '.'));
  const unused = Object.keys(de).filter((k) => !used(k));
  const sure = unused.filter((k) => !quoted.has(k.split('.').pop()));
  const maybe = unused.filter((k) => quoted.has(k.split('.').pop()));
  if (process.argv.includes('--json')) console.log(JSON.stringify({ sure, maybe }, null, 1));
