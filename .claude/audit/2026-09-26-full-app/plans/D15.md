Die Pseudo-Elemente erben Custom Properties vom `html`-Element. Nach dem Build die Set→Arsenal-Transition ansehen. Greift das Token dort nicht, an diesen beiden Stellen das Literal behalten und begründen.
- `feedback-motion.service.ts` (WAAPI-`easing`-String) kann kein CSS-Token lesen und bleibt.

Der Build merkt das nicht. Nur einen Kommentar darüber setzen: `// Same curve as --sc-ease-out (styles.scss); Angular animations hand this string to WAAPI, which cannot read a CSS custom property.` Kein Sammel-`sed` über `src/app`, das diese Zeile mitnimmt.

**Schließt:** AUD-300. Nebeneffekt: `var(--sc-ease-out)` ist minifiziert kürzer als das Literal, das hilft beim Budget in Schritt 12.

```js
// - Oberflächen-/Hebe-Regeln werden an Ort und Stelle eingewickelt (Kaskaden-Reihenfolge bleibt).
// - Gemischte Selektorlisten (".a:hover, .a:focus-visible") werden geteilt: der :hover-Teil
//   wandert in eine Kopie direkt davor, eingewickelt; der Rest bleibt unverändert.
// - Reveal-Regeln (opacity/visibility/display/clip-path/pointer-events/max-height), :is()/:where()
//   und Regeln in @media (hover…)/(pointer: coarse)/(prefers-reduced-motion) bleiben und werden gemeldet.
// - Verschachtelte Regeln (`.nav a { &:hover {…} }`) bekommen das @media im Elternblock;
//   der Angular-Build hebt es korrekt heraus (geprüft auf 0.104.1).
// - Direkt aufeinanderfolgende eingewickelte Regeln teilen sich EINEN @media-Block (Budget).
// styles.scss wird NICHT angefasst – dort von Hand (Schritt 10).
import fs from 'node:fs'; import path from 'node:path'; import { createRequire } from 'node:module';
const require = createRequire(path.resolve('package.json'));
const postcss = require('postcss');
const DRY = process.argv.includes('--dry');
const files = process.argv.slice(2).filter((a) => !a.startsWith('--'));
const REVEAL = /^(opacity|visibility|display|max-height|max-block-size|clip-path|pointer-events)$/;
const MEDIA = '(hover: hover)';
let wrapped = 0; const manual = [];
function blocks(src) {
  const out = []; const re = /styles:\s*\[?\s*`/g; let m;
  while ((m = re.exec(src))) { const s = m.index + m[0].length; const e = src.indexOf('`', s); out.push([s, e]); re.lastIndex = e + 1; }
  return out;
}
function inHoverOrCoarse(node) {
  for (let p = node.parent; p && p.type !== 'root'; p = p.parent)
    if (p.type === 'atrule' && p.name === 'media' && /hover\s*:|pointer:\s*coarse|prefers-reduced-motion/.test(p.params)) return true;
  return false;
}
for (const f of files) {
  let src = fs.readFileSync(f, 'utf8');
  for (const [s, e] of blocks(src).reverse()) {           // von hinten, damit Offsets stimmen
    const root = postcss.parse(src.slice(s, e));
    const targets = [];
    root.walkRules((r) => { if (/:hover/.test(r.selector) && !inHoverOrCoarse(r)) targets.push(r); });
    for (const r of targets) {
      const props = (r.nodes ?? []).filter((n) => n.type === 'decl').map((n) => n.prop);
      const line = src.slice(0, s).split('\n').length + r.source.start.line - 1;
      if (props.some((p) => REVEAL.test(p)) || (r.nodes ?? []).some((n) => n.type === 'rule' || n.type === 'atrule')) {
        manual.push(`${f}:${line}  ${r.selector.replace(/\s+/g, ' ')}`); continue;
      }
      if (r.selector.includes(':is(') || r.selector.includes(':where(')) { manual.push(`${f}:${line}  (is/where) ${r.selector.replace(/\s+/g, ' ')}`); continue; }
      const hov = r.selectors.filter((p) => p.includes(':hover')); const rest = r.selectors.filter((p) => !p.includes(':hover'));
      let hoverRule = r;
      if (rest.length) { hoverRule = r.clone({ selectors: hov }); r.selectors = rest; r.before(hoverRule); }
      const prev = hoverRule.prev();
      if (prev && prev.type === 'atrule' && prev.name === 'media' && prev.params === MEDIA && prev.raws.__ours) {
        prev.append(hoverRule.remove());
      } else {
        const indent = (hoverRule.raws.before ?? '\n').replace(/^[\r\n]+/, '');
        const at = postcss.atRule({ name: 'media', params: MEDIA, raws: { before: hoverRule.raws.before, between: ' ', after: '\n' + indent, __ours: true } });
        hoverRule.replaceWith(at); hoverRule.raws.before = '\n' + indent + '  '; at.append(hoverRule);
      }
      wrapped++;
    }
    root.walkAtRules('media', (a) => { delete a.raws.__ours; });
    src = src.slice(0, s) + root.toString() + src.slice(e);
  }
  if (!DRY) fs.writeFileSync(f, src);
}
console.log(`${wrapped} Regeln eingewickelt${DRY ? ' (dry run)' : ''}.`);
```

  ```js
  #!/usr/bin/env node
  // check-hover-scope.mjs — keep hover effects to devices that can hover.
  //
  // A :hover rule outside `@media (hover: hover)` sticks on touch screens: the
  // tapped card keeps its lift / tint until the next tap somewhere else (audit
  // 2026-09-26, block D15). Allowed without the wrapper:
  //   - rules inside @media (hover: …), (any-hover: …), (pointer: coarse) or
  //     (prefers-reduced-motion: …) — those are the scoping or the resets;
  //   - a rule directly preceded by a `/* hover-ok: <reason> */` comment
  //     (hover reveals that touch users are meant to get by tapping).
  //
  // USAGE
  //   node scripts/check-hover-scope.mjs             # scan src/app/**/*.ts styles + src/styles.scss
  //   node scripts/check-hover-scope.mjs --selftest  # verify the matcher
  import fs from 'node:fs';
  import path from 'node:path';
  import { fileURLToPath } from 'node:url';

  const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
  const SCOPED = /\((?:any-)?hover\s*:|pointer\s*:\s*coarse|prefers-reduced-motion/;

  /** Every `:hover` rule outside a scoping @media and without a hover-ok marker. */
  export function findUnscopedHover(css) {
    const out = [];
    const stack = []; // preludes of open blocks
    let prelude = '';
    let okPending = false;
    let line = 1;
    let preludeLine = 1;
    for (let i = 0; i < css.length; i++) {
      const c = css[i];
      if (c === '\n') line++;
      if (c === '/' && css[i + 1] === '*') {
        const end = css.indexOf('*/', i + 2);
        const body = css.slice(i + 2, end < 0 ? css.length : end);
        if (/hover-ok\s*:/.test(body)) okPending = true;
        line += (body.match(/\n/g) || []).length;
        i = end < 0 ? css.length : end + 1;
        continue;
      }
      if (c === '"' || c === "'") {
        const end = css.indexOf(c, i + 1);
        prelude += c + c; // keep the quotes, drop the content: "x:hover" is not a selector
        i = end < 0 ? css.length : end;
        continue;
      }
      if (c === '{') {
        const p = prelude.trim();
        if (/:hover\b/.test(p) && !p.startsWith('@') && !okPending &&
            !stack.some((s) => s.startsWith('@media') && SCOPED.test(s))) {
          out.push({ line: preludeLine, selector: p.replace(/\s+/g, ' ') });
        }
        stack.push(p);
        prelude = '';
        okPending = false;
        continue;
      }
      if (c === '}') { stack.pop(); prelude = ''; okPending = false; continue; }
      if (c === ';') { prelude = ''; continue; }
      if (!prelude.trim() && !/\s/.test(c)) preludeLine = line;
      prelude += c;
    }
    return out;
  }

  function styleBlocks(src) {
    const out = [];
    const re = /styles:\s*\[?\s*`/g;
    let m;
    while ((m = re.exec(src))) {
      let s = m.index + m[0].length;
      for (;;) {
        const e = src.indexOf('`', s);
        out.push({ css: src.slice(s, e), offset: s });
        const next = src.slice(e + 1).match(/^\s*,\s*`/);
        if (!next) { re.lastIndex = e + 1; break; }
        s = e + 1 + next[0].length;
      }
    }
    return out;
  }

  function walk(dir) {
    return fs.readdirSync(dir, { withFileTypes: true }).flatMap((e) => {
      const p = path.join(dir, e.name);
      return e.isDirectory() ? walk(p) : p.endsWith('.ts') && !p.endsWith('.spec.ts') ? [p] : [];
    });
  }

  function scan() {
    const hits = [];
    for (const f of walk(path.join(ROOT, 'src/app'))) {
      const src = fs.readFileSync(f, 'utf8');
      for (const b of styleBlocks(src)) {
        const base = src.slice(0, b.offset).split('\n').length - 1;
        for (const h of findUnscopedHover(b.css)) hits.push(`${path.relative(ROOT, f)}:${base + h.line}  ${h.selector}`);
      }
    }
    const scss = path.join(ROOT, 'src/styles.scss');
    for (const h of findUnscopedHover(fs.readFileSync(scss, 'utf8'))) hits.push(`src/styles.scss:${h.line}  ${h.selector}`);
    return hits;
  }

  function selftest() {
    const cases = [
      ['.a:hover { color: red; }', 1],
      ['@media (hover: hover) { .a:hover { color: red; } }', 0],
      ['@media (max-width: 600px) { .a:hover { color: red; } }', 1],
      ['@media (hover: hover) and (pointer: fine) { .a:hover { x: y; } }', 0],
      ['@media (prefers-reduced-motion: reduce) { .a:hover { transform: none; } }', 0],
      ['@media (pointer: coarse) { .a:hover { transform: none; } }', 0],
      ['/* hover-ok: reveal, a tap should show it */\n.a:hover .tip { opacity: 1; }', 0],
      ['/* ordinary comment */\n.a:hover { color: red; }', 1],
      ['.a { color: red; } .a:focus-visible { outline: 0; }', 0],
      ['.nav a { &:hover { color: red; } }', 1],
      ['.nav a { @media (hover: hover) { &:hover { color: red; } } }', 0],
      ['.a[title="x:hover"] { color: red; }', 0],
    ];
    let failed = 0;
    for (const [css, want] of cases) {
      const got = findUnscopedHover(css).length;
      const ok = got === want;
      if (!ok) failed++;
      console.log(`${ok ? 'ok  ' : 'FAIL'} ${JSON.stringify(css).slice(0, 70)} → ${got} (want ${want})`);
    }
    console.log(failed ? `\nselftest: ${failed} case(s) failed` : `\nselftest: ${cases.length}/${cases.length} passed`);
    process.exitCode = failed ? 1 : 0;
  }

  if (process.argv.includes('--selftest')) {
    selftest();
  } else {
    const hits = scan();
    if (hits.length) {
      console.error(hits.join('\n'));
      console.error(`\ncheck-hover-scope: ${hits.length} :hover rule(s) outside @media (hover: hover).`);
      console.error('Wrap them, or mark a deliberate touch reveal with /* hover-ok: <reason> */.');
      process.exitCode = 1;
    } else {
      console.log('check-hover-scope: ok');
    }
  }
  ```

  > **Motion stays cheap and mouse-only.** A `:hover` style sits in `@media (hover: hover)` — on a touch screen it would stick after the tap; `npm run check:hover` (prebuild) enforces it, and a tooltip-style reveal a tap should also show is marked `/* hover-ok: <reason> */`. Animate `transform`/`opacity`, never width/height; list transition properties instead of `all`; take curves from `--sc-ease-out` / `--sc-ease-io`; an endless decorative loop gets `scPauseOffscreen`.
